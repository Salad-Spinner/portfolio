// Fetch every product at build time with a native HTTPS stream
const fetchProducts = () => new Promise((resolve, reject) => {
  const req = https.request({
    hostname: 'api.printify.com',
    port: 443,
    path: `/v1/shops/${shopId}/products.json`,
    headers: {
      Authorization: `Bearer ${apiToken}`,
      'Content-Type': 'application/json'
    },
    timeout: 15000
  }, (res) => {
    let data = '';
    res.on('data', (chunk) => { data += chunk; });
    res.on('end', () => resolve(JSON.parse(data)));
  });

  req.on('error', reject);
  req.on('timeout', () => reject(new Error('Printify timed out.')));
  req.end();
});
# Replay guard: never re-trigger the same mail message for a while
def mail_recently_played(msg_id):
    now = time.time()
    last = _recent_mail_ids.get(msg_id, 0)
    if (now - last) < MAIL_REPLAY_GUARD:
        return True
    _recent_mail_ids[msg_id] = now
    if len(_recent_mail_ids) > 100:
        _recent_mail_ids.clear()
    return False

        tft.line(x + 15 - i, y + i, x + 15 + i, y + i, color)
